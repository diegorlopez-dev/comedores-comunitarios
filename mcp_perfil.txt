#### State and Effect
```typescript
import { useState, useEffect } from 'react';
import { supabase } from '../supabase';

const [comedorExistente, setComedorExistente] = useState<{ id: number; nombre: string; barrio: string } | null>(null);

useEffect(() => {
  const fetchComedor = async () => {
    if (user.id) {
      const { data, error } = await supabase
        .from('comedor')
        .select('id, nombre, barrio')
        .eq('usuario_id', user.id)
        .single();

      if (data) {
        setComedorExistente(data);
      } else {
        setComedorExistente(null);
      }
    }
  };

  fetchComedor();
}, [user.id]);
```
